cmake_minimum_required(VERSION 3.20)
project(ItemGlyphs LANGUAGES CXX)

add_library(ncmm_mod SHARED src/item_glyphs.cpp)
target_compile_features(ncmm_mod PRIVATE cxx_std_17)
target_include_directories(ncmm_mod PRIVATE ../../sdk)
target_compile_definitions(ncmm_mod PRIVATE NCMM_MOD_BUILD)

if(MSVC)
  set_property(TARGET ncmm_mod PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(ncmm_mod PRIVATE /EHsc)
endif()

set_target_properties(ncmm_mod PROPERTIES PREFIX "" OUTPUT_NAME "ncmm_mod")

add_executable(ncmm_item_glyphs_test ../../tests/item_glyphs_test.cpp)
target_compile_features(ncmm_item_glyphs_test PRIVATE cxx_std_17)
target_include_directories(ncmm_item_glyphs_test PRIVATE ../../host_patch)
if(MSVC)
  set_property(TARGET ncmm_item_glyphs_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(ncmm_item_glyphs_test PRIVATE /EHsc)
endif()

set(NCMM_CDDA_SOURCE "" CACHE PATH "CDDA 0546 source for original wcwidth validation")
if(NCMM_CDDA_SOURCE)
  file(SHA256 "${NCMM_CDDA_SOURCE}/src/wcwidth.cpp" NCMM_WCWIDTH_SHA)
  if(NOT NCMM_WCWIDTH_SHA STREQUAL "859d21c69ad9ce1c33e7549beead830c7859089d1cee998eaaafc5020068c12a")
    message(FATAL_ERROR "Expected wcwidth.cpp from CDDA e262adb299a7613b4aedc5f12c08fe0413c56a84")
  endif()
  add_library(ncmm_item_glyphs_cdda_compile OBJECT ../../tests/item_glyphs_cdda_compile.cpp)
  target_compile_features(ncmm_item_glyphs_cdda_compile PRIVATE cxx_std_17)
  target_include_directories(ncmm_item_glyphs_cdda_compile PRIVATE ../../host_patch
    "${NCMM_CDDA_SOURCE}/src" "${NCMM_CDDA_SOURCE}/src/third-party")
  target_sources(ncmm_item_glyphs_test PRIVATE "${NCMM_CDDA_SOURCE}/src/wcwidth.cpp")
  target_include_directories(ncmm_item_glyphs_test PRIVATE "${NCMM_CDDA_SOURCE}/src")
  target_compile_definitions(ncmm_item_glyphs_test PRIVATE NCMM_TEST_CDDA_WIDTH)
endif()
enable_testing()
add_test(NAME item_glyphs_classifier COMMAND ncmm_item_glyphs_test)
