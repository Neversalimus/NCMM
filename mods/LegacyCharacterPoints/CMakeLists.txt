cmake_minimum_required(VERSION 3.20)
project(LegacyCharacterPoints LANGUAGES CXX)
add_library(ncmm_mod SHARED src/legacy_character_points.cpp)
target_compile_features(ncmm_mod PRIVATE cxx_std_17)
target_include_directories(ncmm_mod PRIVATE ../../sdk)
target_compile_definitions(ncmm_mod PRIVATE NCMM_MOD_BUILD)
set_target_properties(ncmm_mod PROPERTIES PREFIX "" OUTPUT_NAME "ncmm_mod")
add_executable(ncmm_legacy_points_test tests/legacy_points_test.cpp)
target_compile_features(ncmm_legacy_points_test PRIVATE cxx_std_17)
target_include_directories(ncmm_legacy_points_test PRIVATE ../../host_patch ../../sdk)
add_executable(ncmm_legacy_points_module_test tests/module_test.cpp src/legacy_character_points.cpp)
target_compile_features(ncmm_legacy_points_module_test PRIVATE cxx_std_17)
target_include_directories(ncmm_legacy_points_module_test PRIVATE ../../sdk)
target_compile_definitions(ncmm_legacy_points_module_test PRIVATE NCMM_MOD_BUILD)
if(MSVC)
  foreach(t ncmm_mod ncmm_legacy_points_test ncmm_legacy_points_module_test)
    set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    # Exported locale callbacks report failure to the Host exception boundary.
    target_compile_options(${t} PRIVATE /EHsc- /utf-8)
  endforeach()
endif()
enable_testing()
add_test(NAME classic_accounting COMMAND ncmm_legacy_points_test)
add_test(NAME module_lifecycle COMMAND ncmm_legacy_points_module_test)
