cmake_minimum_required(VERSION 3.20)
project(FirstPersonView LANGUAGES CXX)
find_package(Threads REQUIRED)

add_library(ncmm_mod SHARED src/module.cpp)
target_compile_definitions(ncmm_mod PRIVATE NCMM_MOD_BUILD)
target_include_directories(ncmm_mod PRIVATE ../../sdk ../graphics_bridge)
target_compile_features(ncmm_mod PRIVATE cxx_std_17)
set_target_properties(ncmm_mod PROPERTIES PREFIX "" OUTPUT_NAME "ncmm_mod")
add_executable(first_person_tests tests/renderer_test.cpp src/module.cpp)
add_executable(first_person_hybrid_tests tests/hybrid_test.cpp)
add_executable(first_person_hybrid_module_tests tests/hybrid_module_test.cpp src/module.cpp)
target_include_directories(first_person_hybrid_module_tests PRIVATE ../../sdk ../graphics_bridge src)
target_compile_features(first_person_hybrid_module_tests PRIVATE cxx_std_17)
target_compile_definitions(first_person_hybrid_module_tests PRIVATE NCMM_MOD_BUILD)
target_include_directories(first_person_hybrid_tests PRIVATE ../graphics_bridge src)
target_compile_features(first_person_hybrid_tests PRIVATE cxx_std_17)
target_include_directories(first_person_tests PRIVATE ../../sdk ../graphics_bridge src)
target_compile_features(first_person_tests PRIVATE cxx_std_17)
target_compile_definitions(first_person_tests PRIVATE NCMM_MOD_BUILD)
foreach(t ncmm_mod first_person_tests first_person_hybrid_tests first_person_hybrid_module_tests)
  target_link_libraries(${t} PRIVATE Threads::Threads)
endforeach()
if(MSVC)
  foreach(t ncmm_mod first_person_tests first_person_hybrid_tests first_person_hybrid_module_tests)
    set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(${t} PRIVATE /W4 /utf-8)
  endforeach()
endif()
enable_testing()
add_test(NAME first_person_contract COMMAND first_person_tests)
add_test(NAME first_person_hybrid COMMAND first_person_hybrid_tests)
add_test(NAME first_person_hybrid_module COMMAND first_person_hybrid_module_tests)

add_executable(first_person_presentation_tests tests/presentation_test.cpp)
target_include_directories(first_person_presentation_tests PRIVATE ../graphics_bridge src)
target_compile_features(first_person_presentation_tests PRIVATE cxx_std_17)
target_link_libraries(first_person_presentation_tests PRIVATE Threads::Threads)
if(MSVC)
  set_property(TARGET first_person_presentation_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(first_person_presentation_tests PRIVATE /W4 /utf-8)
endif()
add_test(NAME first_person_presentation COMMAND first_person_presentation_tests)

option(FIRST_PERSON_BUILD_BENCHMARK "Build the manual camera/composition benchmark" OFF)
if(FIRST_PERSON_BUILD_BENCHMARK)
  add_executable(first_person_benchmark tests/renderer_benchmark.cpp)
  target_include_directories(first_person_benchmark PRIVATE ../graphics_bridge src)
  target_compile_features(first_person_benchmark PRIVATE cxx_std_17)
  target_link_libraries(first_person_benchmark PRIVATE Threads::Threads)
  if(MSVC)
    set_property(TARGET first_person_benchmark PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(first_person_benchmark PRIVATE /W4 /utf-8)
  endif()
endif()
