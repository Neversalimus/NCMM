cmake_minimum_required(VERSION 3.20)
project(FirstPersonView LANGUAGES CXX)
find_package(Threads REQUIRED)

add_library(first_person_build_options INTERFACE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/mod.json" first_person_manifest)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/mod.json"
  "${CMAKE_CURRENT_SOURCE_DIR}/assets/psx_arms/manifest.json"
  "${CMAKE_CURRENT_SOURCE_DIR}/assets/psx_arms/model.generated.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/assets/psx_arms/bare.rgba"
  "${CMAKE_CURRENT_SOURCE_DIR}/assets/psx_arms/gloves.rgba")
string(JSON first_person_version GET "${first_person_manifest}" version)
target_compile_definitions(first_person_build_options INTERFACE FIRST_PERSON_VERSION="${first_person_version}")
target_include_directories(first_person_build_options INTERFACE ../../sdk ../graphics_bridge src)
target_compile_features(first_person_build_options INTERFACE cxx_std_17)
target_link_libraries(first_person_build_options INTERFACE Threads::Threads)
if(MSVC)
  target_compile_options(first_person_build_options INTERFACE /W4 /utf-8)
endif()
option(FIRST_PERSON_SANITIZE "Instrument native modules and tests with ASan/UBSan" OFF)
if(FIRST_PERSON_SANITIZE AND NOT MSVC)
  target_compile_options(first_person_build_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(first_person_build_options INTERFACE -fsanitize=address,undefined)
endif()
function(first_person_target target)
  target_link_libraries(${target} PRIVATE first_person_build_options)
  if(NOT target STREQUAL "first_person_support")
    target_link_libraries(${target} PRIVATE first_person_support)
  endif()
  if(MSVC)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endfunction()

set(first_person_psx_assets "${CMAKE_CURRENT_SOURCE_DIR}/assets/psx_arms")
file(READ "${first_person_psx_assets}/manifest.json" first_person_psx_manifest)
set(first_person_psx_texture_cpp "#include \"viewmodel_asset.hpp\"\nnamespace first_person::viewmodel {\n")
set(first_person_psx_texture_index 0)
foreach(variant bare gloves)
  file(SHA256 "${first_person_psx_assets}/${variant}.rgba" actual_hash)
  string(JSON expected_hash GET "${first_person_psx_manifest}" textures ${first_person_psx_texture_index} rgba8_sha256)
  if(NOT actual_hash STREQUAL expected_hash)
    message(FATAL_ERROR "PSX ${variant} pixels differ from the validated import")
  endif()
  file(READ "${first_person_psx_assets}/${variant}.rgba" texture_hex HEX)
  string(REGEX REPLACE "(........)" "0x\\1u,\n" texture_words "${texture_hex}")
  string(REGEX REPLACE "s$" "" texture_symbol "${variant}")
  string(APPEND first_person_psx_texture_cpp "const uint32_t psx_${texture_symbol}_pixels[512*512]={\n${texture_words}};\n")
  math(EXPR first_person_psx_texture_index "${first_person_psx_texture_index}+1")
endforeach()
string(APPEND first_person_psx_texture_cpp "}\n")
file(SHA256 "${first_person_psx_assets}/model.generated.cpp" actual_hash)
string(JSON expected_hash GET "${first_person_psx_manifest}" generated_cpp_sha256)
if(NOT actual_hash STREQUAL expected_hash)
  message(FATAL_ERROR "PSX model tables differ from the validated import")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/psx_arms_textures.cpp" CONTENT "${first_person_psx_texture_cpp}")
set(first_person_held_assets "${CMAKE_CURRENT_SOURCE_DIR}/assets/held_items/generated")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${first_person_held_assets}/manifest.json")
file(READ "${first_person_held_assets}/manifest.json" first_person_held_manifest)
string(JSON first_person_held_schema GET "${first_person_held_manifest}" schema)
string(JSON first_person_held_count LENGTH "${first_person_held_manifest}" files)
if(NOT first_person_held_schema EQUAL 1 OR NOT first_person_held_count EQUAL 32)
  message(FATAL_ERROR "Unsupported held model inventory")
endif()
set(first_person_held_sources)
math(EXPR first_person_held_last "${first_person_held_count}-1")
foreach(index RANGE ${first_person_held_last})
  string(JSON source_name MEMBER "${first_person_held_manifest}" files ${index})
  if(NOT source_name MATCHES "^[a-z_]+[.]generated[.]cpp$")
    message(FATAL_ERROR "Unsafe held model source path")
  endif()
  string(JSON expected_hash GET "${first_person_held_manifest}" files "${source_name}")
  file(SHA256 "${first_person_held_assets}/${source_name}" actual_hash)
  if(NOT actual_hash STREQUAL expected_hash)
    message(FATAL_ERROR "Held resource ${source_name} differs from the validated offline import")
  endif()
  list(APPEND first_person_held_sources "${first_person_held_assets}/${source_name}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${first_person_held_assets}/${source_name}")
endforeach()
add_library(first_person_held_tables OBJECT ${first_person_held_sources})
target_link_libraries(first_person_held_tables PRIVATE first_person_build_options)
set_target_properties(first_person_held_tables PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set_property(TARGET first_person_held_tables PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(first_person_support STATIC src/creature_appearance_debug.cpp src/viewmodel_animation.cpp src/viewmodel.cpp
  src/held_item.cpp $<TARGET_OBJECTS:first_person_held_tables>
  assets/psx_arms/model.generated.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/psx_arms_textures.cpp")
first_person_target(first_person_support)
set_target_properties(first_person_support PROPERTIES POSITION_INDEPENDENT_CODE ON)
