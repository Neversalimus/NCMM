cmake_minimum_required(VERSION 3.20)
project(FirstPersonView LANGUAGES CXX)
find_package(Threads REQUIRED)

add_library(first_person_build_options INTERFACE)
target_include_directories(first_person_build_options INTERFACE ../../sdk ../graphics_bridge src)
target_compile_features(first_person_build_options INTERFACE cxx_std_17)
target_link_libraries(first_person_build_options INTERFACE Threads::Threads)
if(MSVC)
  target_compile_options(first_person_build_options INTERFACE /W4 /utf-8)
endif()
option(FIRST_PERSON_SANITIZE "Instrument native modules and tests with ASan/UBSan" OFF)
if(FIRST_PERSON_SANITIZE AND NOT MSVC)
  target_compile_options(first_person_build_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(first_person_build_options INTERFACE -fsanitize=address,undefined)
endif()
function(first_person_target target)
  target_link_libraries(${target} PRIVATE first_person_build_options)
  if(MSVC)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endfunction()

add_library(ncmm_mod SHARED src/module.cpp)
target_compile_definitions(ncmm_mod PRIVATE NCMM_MOD_BUILD)
set_target_properties(ncmm_mod PROPERTIES PREFIX "" OUTPUT_NAME "ncmm_mod")
first_person_target(ncmm_mod)
enable_testing()
add_executable(first_person_multilevel_render_tests tests/multilevel_render_test.cpp)
first_person_target(first_person_multilevel_render_tests)
add_test(NAME first_person_multilevel_render COMMAND first_person_multilevel_render_tests)
add_executable(first_person_multilevel_gpu_cache_tests tests/multilevel_gpu_cache_test.cpp)
first_person_target(first_person_multilevel_gpu_cache_tests)
if(FIRST_PERSON_SANITIZE AND NOT MSVC)
  # The old discarded dense-grid hash may be optimized away before ASan.
  # Exercise sparse reads without allowing that optimization to hide a bug.
  target_compile_options(first_person_multilevel_gpu_cache_tests PRIVATE -O0)
endif()
add_test(NAME first_person_multilevel_gpu_cache COMMAND first_person_multilevel_gpu_cache_tests)
foreach(pair "camera_motion|camera_motion_test" "contract|renderer_test" "hybrid|hybrid_test" "hybrid_module|hybrid_module_test" "presentation|presentation_test" "distance|distance_test" "optimization|optimization_test" "performance_header|performance_header_test" "creature_semantics|creature_semantics_test" "creature_mesh|creature_mesh_test" "creature_appearance|creature_appearance_test" "creature_animation|creature_animation_test" "creature_lod|creature_lod_test" "creature_cache|creature_cache_test" "creature_render|creature_render_test" "creature_module|creature_module_test" "gpu_contract|gpu_contract_test" "gpu_module|gpu_module_test" "facade_continuity|facade_continuity_test" "multilevel_policy|multilevel_policy_test" "multilevel_cache|multilevel_cache_test" "multilevel_module|multilevel_module_test")
  string(REPLACE "|" ";" fields "${pair}")
  list(GET fields 0 name)
  list(GET fields 1 source)
  if(name STREQUAL "contract")
    set(target first_person_tests)
  else()
    set(target first_person_${name}_tests)
  endif()
  add_executable(${target} tests/${source}.cpp)
  first_person_target(${target})
  add_test(NAME first_person_${name} COMMAND ${target})
endforeach()
option(FIRST_PERSON_BUILD_GPU_TESTS "Qualify actual shared-device SDL_GPU geometry and UI" OFF)

first_person_target(first_person_gpu_geometry_tests)
add_test(NAME first_person_gpu_geometry COMMAND first_person_gpu_geometry_tests)
