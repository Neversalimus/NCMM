cmake_minimum_required(VERSION 3.20)
project(FirstPersonView LANGUAGES CXX)
find_package(Threads REQUIRED)

add_library(first_person_build_options INTERFACE)
target_include_directories(first_person_build_options INTERFACE ../../sdk ../graphics_bridge src)
target_compile_features(first_person_build_options INTERFACE cxx_std_17)
target_link_libraries(first_person_build_options INTERFACE Threads::Threads)
if(MSVC)
  target_compile_options(first_person_build_options INTERFACE /W4 /utf-8)
endif()
option(FIRST_PERSON_SANITIZE "Instrument native modules and tests with ASan/UBSan" OFF)
if(FIRST_PERSON_SANITIZE AND NOT MSVC)
  target_compile_options(first_person_build_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(first_person_build_options INTERFACE -fsanitize=address,undefined)
endif()
function(first_person_target target)
  target_link_libraries(${target} PRIVATE first_person_build_options)
  if(MSVC)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endfunction()

add_library(ncmm_mod SHARED src/module.cpp)
target_compile_definitions(ncmm_mod PRIVATE NCMM_MOD_BUILD)
set_target_properties(ncmm_mod PROPERTIES PREFIX "" OUTPUT_NAME "ncmm_mod")
first_person_target(ncmm_mod)
enable_testing()
foreach(pair "contract|renderer_test" "hybrid|hybrid_test" "hybrid_module|hybrid_module_test" "presentation|presentation_test" "distance|distance_test" "optimization|optimization_test")
  string(REPLACE "|" ";" fields "${pair}")
  list(GET fields 0 name)
  list(GET fields 1 source)
  if(name STREQUAL "contract")
    set(target first_person_tests)
  else()
    set(target first_person_${name}_tests)
  endif()
  add_executable(${target} tests/${source}.cpp)
  first_person_target(${target})
  add_test(NAME first_person_${name} COMMAND ${target})
endforeach()
# ABI/lifecycle tests now exercise the exact shared library that is shipped,
# rather than compiling a separate copy of module.cpp into each test binary.
foreach(target first_person_tests first_person_hybrid_module_tests)
  target_link_libraries(${target} PRIVATE ncmm_mod)
endforeach()

option(FIRST_PERSON_BUILD_BENCHMARK "Build the manual camera/composition benchmark" OFF)
set(FIRST_PERSON_BENCHMARK_ROUNDS 5 CACHE STRING "Rounds for the non-gating timing artifact")
if(FIRST_PERSON_BUILD_BENCHMARK)
  add_executable(first_person_benchmark tests/renderer_benchmark.cpp)
  first_person_target(first_person_benchmark)
  add_custom_target(first_person_measure
    COMMAND $<TARGET_FILE:first_person_benchmark> ${FIRST_PERSON_BENCHMARK_ROUNDS} distance ${CMAKE_BINARY_DIR}/first-person-benchmark.json
    DEPENDS first_person_benchmark
    COMMENT "Record reference/optimized timings without an FPS pass/fail threshold")
endif()
